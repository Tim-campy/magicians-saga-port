function(project_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
  endif()
endfunction()

option(SAGA_BUILD_VIEWER "Build the optional desktop PNG research viewer" OFF)
if(SAGA_BUILD_VIEWER)
  set(FETCHCONTENT_BASE_DIR "${CMAKE_SOURCE_DIR}/build/_deps" CACHE PATH "Local downloaded source dependencies")
  include(FetchContent)
  if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
  endif()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3
    URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz
    URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3)
  FetchContent_MakeAvailable(SDL3)
  set(STB_DIR "${CMAKE_SOURCE_DIR}/build/_deps/stb")
  set(STB_SHA256 594c2fe35d49488b4382dbfaec8f98366defca819d916ac95becf3e75f4200b3)
  file(MAKE_DIRECTORY "${STB_DIR}")
  set(STB_ACTUAL "missing")
  if(EXISTS "${STB_DIR}/stb_image.h")
    file(SHA256 "${STB_DIR}/stb_image.h" STB_ACTUAL)
  endif()
  if(NOT STB_ACTUAL STREQUAL STB_SHA256)
    file(DOWNLOAD
      https://raw.githubusercontent.com/nothings/stb/003501f6d7a8668ce8b6d875e34dd97ad0d5d5b5/stb_image.h
      "${STB_DIR}/stb_image.h" EXPECTED_HASH SHA256=${STB_SHA256} TLS_VERIFY ON)
  endif()
  add_executable(asset_viewer src/platform/desktop/asset_viewer.cpp)
  target_compile_features(asset_viewer PRIVATE cxx_std_17)
  target_include_directories(asset_viewer SYSTEM PRIVATE "${STB_DIR}")
  target_link_libraries(asset_viewer PRIVATE SDL3::SDL3-static opt_grid)
  project_warnings(asset_viewer)
endif()
